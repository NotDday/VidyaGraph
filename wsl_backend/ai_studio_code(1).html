from sklearn.svm import SVC
from sklearn.tree import DecisionTreeClassifier

base_models = [
    ('rf', RandomForestClassifier(n_estimators=200, class_weight='balanced', random_state=42)),
    ('svm', SVC(kernel='rbf', C=10, probability=True, class_weight='balanced', random_state=42)),

]

# --- ANN BACKPROPAGATION IN PURE NUMPY (IF REQUIRED WITHOUT TENSORFLOW) ---
import numpy as np

def sigmoid(x):
    return 1 / (1 + np.exp(-np.clip(x, -500, 500)))

def derivatives_sigmoid(x):
    return x * (1 - x)

X_nn = np.array(X_train, dtype=float)
y_nn = np.array(y_train, dtype=float).reshape(-1, 1)

epoch = 5000
lr = 0.1
inputlayer_neurons = X_nn.shape[1]
hiddenlayer_neurons = 8
output_neurons = 1

np.random.seed(42)
wh = np.random.uniform(size=(inputlayer_neurons, hiddenlayer_neurons))
bh = np.random.uniform(size=(1, hiddenlayer_neurons))
wout = np.random.uniform(size=(hiddenlayer_neurons, output_neurons))
bout = np.random.uniform(size=(1, output_neurons))

for i in range(epoch):
    hinp = np.dot(X_nn, wh) + bh
    hlayer_act = sigmoid(hinp)
    outinp = np.dot(hlayer_act, wout) + bout
    output = sigmoid(outinp)
    
    EO = y_nn - output
    d_output = EO * derivatives_sigmoid(output)
    EH = d_output.dot(wout.T)
    d_hiddenlayer = EH * derivatives_sigmoid(hlayer_act)
    
    wout += hlayer_act.T.dot(d_output) * lr
    bout += np.sum(d_output, axis=0, keepdims=True) * lr
    wh += X_nn.T.dot(d_hiddenlayer) * lr
    bh += np.sum(d_hiddenlayer, axis=0, keepdims=True) * lr

hlayer_test = sigmoid(np.dot(np.array(X_test, dtype=float), wh) + bh)
y_pred_prob = sigmoid(np.dot(hlayer_test, wout) + bout).flatten()
y_pred = (y_pred_prob > 0.5).astype(int)

## Syllabus Model Hyperparameter Quick Reference

              

                
                  

                    Model
                    Key Parameters
                    Recommended Exam Setting
                  

                
                
                  

                    LinearRegression / Ridge
                    fit_intercept=True, alpha=1.0
                    Use RobustScaler + TransformedTargetRegressor for skewed targets.
                  

                  

                    LogisticRegression
                    C=1.0, max_iter=1000, class_weight='balanced'
                    Set max_iter=1000 to avoid convergence warnings.
                  

                  

                    SVC / SVR (SVM)
                    kernel='rbf', C=10.0, gamma='scale', probability=True
                    For SVR on large targets (prices), use C=100 or 1000.
                  

                  

                    DecisionTree (Clf/Reg)
                    max_depth=7, min_samples_split=5
                    Constrain max_depth=5..10 so the tree does not overfit noise.
                  

                  

                    RandomForest (Clf/Reg)
                    n_estimators=200, max_depth=12, class_weight='balanced'
                    Best single tabular model out-of-the-box.
                  

                  

                    tf.keras.models.Sequential (ANN)
                    Dense(64, activation='relu'), optimizer='adam', epochs=60
                    Output layer: 'sigmoid' (binary), 'softmax' (multiclass), 'linear' (regression).
